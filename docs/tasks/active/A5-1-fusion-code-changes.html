A5-1 능력 합성(Fusion) — 수정·추가 코드 상세 보고서

구현 보고서 배선 완료 코드 변경 상세 · 2026-07-13

1. 작업 요지

한 타워(캐릭터)마다 합성 계보(족보)를 두고, 재료 능력 2개가 각각 최대 레벨(MAX)에 도달하면 레벨업 카드 선택지에 합성 카드(Fusion)를 제시한다. 선택하면 재료 2개를 소진하고 결과 능력을 얻는다.

설계 원칙은 데이터·로직 분리(SRP)다. 무엇을 합성하는가(계보)는 순수 데이터(ScriptableObject)로, 지금 합성 가능한가·어떻게 적용하는가는 뷰 비의존 순수 로직으로 나눴다.

이번 보고서는 신규 2파일 + 수정 8파일 + 테스트 2파일의 변경 내용 전부를 코드로 나열한다. 마지막 수정은 Presenter의 배선 누락(계보 미전달)을 복구한 2줄로, 이걸로 인게임에서 합성 카드가 실제로 뜬다.

3. 신규 파일 — 데이터·판정 로직

(1) FusionRecipeSet.cs — 계보 데이터 신규

한 타워의 합성 레시피 목록을 담는 ScriptableObject. FusionRecipe 구조체(재료A·재료B·결과)의 리스트만 갖는 순수 데이터 컨테이너이며, 판정·적용 로직은 일절 포함하지 않는다(SRP). OnValidate는 디자이너의 실수 (참조 누락·자기합성·결과가 재료와 동일)를 에디터 콘솔 경고로만 알린다.

Assets/Scripts/Systems/Cards/FusionRecipeSet.cs
/// 한 타워(캐릭터)의 합성 계보 — 재료 2개→결과 레시피 목록 (데이터만)
[CreateAssetMenu(fileName = "FusionRecipeSet", menuName = "DefenseDot/Fusion Recipe Set")]
public sealed class FusionRecipeSet : ScriptableObject
{
    public List<FusionRecipe> recipes = new List<FusionRecipe>();   // 계보 = 레시피 목록

    /// 디자이너 실수(null·자기합성·결과=재료)를 콘솔 경고로 알림
    private void OnValidate()
    {
        if (recipes == null) return;
        for (int i = 0; i < recipes.Count; i++)
        {
            FusionRecipe r = recipes[i];
            if (r.materialA == null || r.materialB == null || r.result == null)
                Debug.LogWarning($"[FusionRecipeSet] {name} 레시피 {i}: 참조 누락", this);
            else if (r.materialA == r.materialB)
                Debug.LogWarning($"[FusionRecipeSet] {name} 레시피 {i}: 재료 A==B", this);
            else if (r.result == r.materialA || r.result == r.materialB)
                Debug.LogWarning($"[FusionRecipeSet] {name} 레시피 {i}: 결과가 재료와 같음", this);
        }
    }
}

/// 합성 레시피 1건 — 재료 2개 소진 → 결과 1개
[System.Serializable]
public struct FusionRecipe
{
    public AbilityData materialA;   // 재료 A
    public AbilityData materialB;   // 재료 B
    public AbilityData result;      // 결과(일반 카드 풀 제외)
}
  • 레시피는 능력 3개의 참조만 담는다 — 로직 없음. 계보를 늘리려면 이 리스트에 항목만 추가하면 된다(OCP).
  • result는 일반 카드 풀에 넣지 않는 능력을 가리키는 것이 원칙 — 합성으로만 얻게 하기 위함.

(2) FusionResolver.cs — 가용 합성 판정 신규

"지금 이 로드아웃에서 만들 수 있는 합성"을 판정하는 뷰 비의존 순수 로직. 세 조건을 모두 만족하는 레시피만 반환한다: ① 재료 2개를 모두 보유 ② 둘 다 MAX 레벨 ③ 결과는 아직 미보유.

Assets/Scripts/Systems/Cards/FusionResolver.cs
public sealed class FusionResolver
{
    /// 계보에서 지금 만들 수 있는 합성 목록을 반환
    public List<AvailableFusion> Available(AbilityLoadout loadout, FusionRecipeSet lineage)
    {
        var result = new List<AvailableFusion>();
        if (loadout == null || lineage == null || lineage.recipes == null) return result;

        for (int i = 0; i < lineage.recipes.Count; i++)
        {
            FusionRecipe r = lineage.recipes[i];
            if (r.materialA == null || r.materialB == null || r.result == null) continue;
            if (loadout.Contains(r.result)) continue;   // ③ 결과 미보유 게이트

            AbilityInstance a = FindMaxed(loadout, r.materialA);   // ①② 보유+MAX
            AbilityInstance b = FindMaxed(loadout, r.materialB);
            if (a == null || b == null) continue;

            result.Add(new AvailableFusion(r, a, b));
        }
        return result;
    }

    /// 로드아웃에서 해당 능력의 MAX 인스턴스를 찾음(없거나 비MAX면 null)
    private static AbilityInstance FindMaxed(AbilityLoadout loadout, AbilityData data)
    {
        AbilityInstance inst = Find(loadout.Actives, data);
        if (inst == null) inst = Find(loadout.Passives, data);
        if (inst == null || inst.level < inst.data.maxLevel) return null;
        return inst;
    }
    // Find(list, data): 목록에서 설계도 일치 인스턴스 탐색
}

반환 타입 AvailableFusion은 레시피 + 소진할 재료 인스턴스 2개를 묶은 readonly struct다. 재료 인스턴스(레벨·상태를 가진 실제 슬롯 객체)를 함께 실어 보내, 적용 단계에서 정확히 그 인스턴스를 제거하게 한다.
